// Spike 2 throwaway: stand-in for deparked-admin. Not product code.
const VERSION = "__VERSION__";
const CORS = {
  "access-control-allow-origin": "*",
  "access-control-allow-headers": "content-type",
  "access-control-allow-methods": "GET,POST,OPTIONS",
};
const json = (o, status = 200) =>
  new Response(JSON.stringify(o, null, 2), { status, headers: { ...CORS, "content-type": "application/json" } });

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    if (req.method === "OPTIONS") return new Response(null, { headers: CORS });

    if (url.pathname === "/health") {
      const row = await env.DB.prepare("select count(*) as n from sqlite_master").first();
      return json({
        version: VERSION,
        d1: row !== null,
        hasToken: typeof env.CF_TOKEN === "string" && env.CF_TOKEN.length > 20,
        hasSetupSecret: typeof env.SETUP_SECRET === "string" && env.SETUP_SECRET.length > 20,
        bindings: Object.keys(env).sort(),
      });
    }

    // PBKDF2-SHA256, the hash the real setup/login will use. CPU time is read from analytics afterwards.
    if (url.pathname === "/hash" && req.method === "POST") {
      const iterations = Math.min(Number(url.searchParams.get("i") || 100000), 100000);
      const salt = crypto.getRandomValues(new Uint8Array(16));
      const key = await crypto.subtle.importKey("raw", new TextEncoder().encode("correct horse battery staple"), "PBKDF2", false, ["deriveBits"]);
      const bits = await crypto.subtle.deriveBits({ name: "PBKDF2", hash: "SHA-256", salt, iterations }, key, 256);
      return json({ iterations, bytes: bits.byteLength });
    }

    // Self-update: download new code, PUT our own script keeping bindings and secrets.
    if (url.pathname === "/update" && req.method === "POST") {
      const { secret } = await req.json().catch(() => ({}));
      if (secret !== env.SETUP_SECRET) return json({ error: "bad secret" }, 403);
      let n = 0;
      const f = (u, i) => (n++, fetch(u, i));
      const src = await (await f(env.RELEASE_URL)).text();
      const code = src.replace("__VERSION__", "2");
      const metadata = {
        main_module: "admin.js",
        compatibility_date: "2026-10-01",
        bindings: [],
        keep_bindings: ["secret_text", "d1", "plain_text"],
        observability: { enabled: true },
      };
      const form = new FormData();
      form.append("metadata", new Blob([JSON.stringify(metadata)], { type: "application/json" }));
      form.append("admin.js", new Blob([code], { type: "application/javascript+module" }), "admin.js");
      const r = await f(`https://api.cloudflare.com/client/v4/accounts/${env.ACCOUNT_ID}/workers/scripts/${env.SCRIPT_NAME}`, {
        method: "PUT",
        headers: { authorization: `Bearer ${env.CF_TOKEN}` },
        body: form,
      });
      const j = await r.json();
      return json({ ok: j.success, errors: j.errors, deployedVersion: "2", subrequests: n });
    }

    return json({ error: "not found", version: VERSION }, 404);
  },
};
